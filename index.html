// Audio player integration
class LyricsPlayer {
  async fetch(track) {
    const u = new URL(API);
    if (track.videoId) u.searchParams.set("videoId", track.videoId);
    else {
      u.searchParams.set("title", track.title);
      u.searchParams.set("artist", track.artist);
      u.searchParams.set("source", track.source || "spotify");
    }
    return await fetch(u).then(r => r.json());
  }

  parseLrc(lrc) {
    return lrc.split("\n").reduce((acc, line) => {
      const m = line.match(/\[(\d+):(\d+)\.(\d+)\](.*)/);
      if (m) {
        const [, mm, ss, cs, text] = m;
        acc.push({ time: +mm * 60 + +ss + +cs / 100, text });
      }
      return acc;
    }, []);
  }
}

# Parse LRC
import re
def parse_lrc(lrc):
    lines = []
    for line in lrc.split("\n"):
        m = re.match(r"\[(\d+):(\d+)\.(\d+)\](.*)", line)
        if m:
            mm, ss, cs, text = m.groups()
            lines.append({
                "time": int(mm) * 60 + int(ss) + int(cs) / 100,
                "text": text
            })
    return lines
